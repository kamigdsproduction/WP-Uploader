async function publishToWordPress() { const url = document.getElementById('wpUrl').value.replace(/\/$/, ""); const user = document.getElementById('wpUser').value; const pass = document.getElementById('wpPass').value; const title = document.getElementById('postTitle').value; const imageFile = document.getElementById('postImage').files[0]; const status = document.getElementById('postStatus').value; const log = document.getElementById('logArea'); const contentHtml = quill.root.innerHTML; if (!url || !user || !pass || !title) { alert("Harap isi URL, Username, Password, dan Judul Artikel!"); return; } log.classList.remove('hidden', 'bg-red-50', 'bg-green-50', 'text-red-700', 'text-green-700'); log.classList.add('bg-blue-50', 'text-blue-700'); log.innerText = "Memproses pengiriman..."; const authHeader = 'Basic ' + btoa(user + ':' + pass); let mediaId = null; try { // Step 1: Upload Gambar (Menggunakan format ?rest_route=) if (imageFile) { log.innerText = "Mengunggah gambar ke WordPress..."; const formData = new FormData(); formData.append('file', imageFile); const imgRes = await fetch(`${url}/?rest_route=/wp/v2/media`, { method: 'POST', headers: { 'Authorization': authHeader }, body: formData }); if (!imgRes.ok) { const errData = await imgRes.json().catch(() => ({})); throw new Error(errData.message || `Gagal upload gambar (Status ${imgRes.status})`); } const imgData = await imgRes.json(); mediaId = imgData.id; } // Step 2: Publish Artikel (Menggunakan format ?rest_route=) log.innerText = "Memuat postingan ke WordPress..."; const postPayload = { title: title, content: contentHtml, status: status, featured_media: mediaId }; const postRes = await fetch(`${url}/?rest_route=/wp/v2/posts`, { method: 'POST', headers: { 'Authorization': authHeader, 'Content-Type': 'application/json' }, body: JSON.stringify(postPayload) }); if (!postRes.ok) { const errData = await postRes.json().catch(() => ({})); throw new Error(errData.message || `Gagal mempublikasikan artikel (Status ${postRes.status})`); } const postData = await postRes.json(); log.classList.remove('bg-blue-50', 'text-blue-700'); log.classList.add('bg-green-50', 'text-green-700'); log.innerHTML = `Berhasil! Artikel telah diterbitkan.
Buka Artikel (${postData.link})`; } catch (err) { log.classList.remove('bg-blue-50', 'text-blue-700'); log.classList.add('bg-red-50', 'text-red-700'); log.innerText = "Terjadi Kesalahan: " + err.message; } }
